const playerName = localStorage.getItem("playerName"); const text = document.getElementById("storyText"); const button = document.getElementById("nextBtn"); const scenes = [ `Welcome ${playerName}...`, "You have discovered a mysterious museum.", "A hidden Time Portal has awakened.", "The timeline has been broken.", "Collect the Five Keys of Time.", "Restore history and return home." ]; let sceneIndex = 0; let charIndex = 0; function typeScene(){ if(charIndex < scenes[sceneIndex].length){ text.innerHTML += scenes[sceneIndex].charAt(charIndex); charIndex++; setTimeout(typeScene,40); } else{ setTimeout(nextScene,1500); } } function nextScene(){ sceneIndex++; charIndex = 0; if(sceneIndex >= scenes.length){ button.style.display = "inline-block"; return; } text.innerHTML = ""; typeScene(); } typeScene(); button.onclick = function(){ window.location.href = "portal.html"; };
